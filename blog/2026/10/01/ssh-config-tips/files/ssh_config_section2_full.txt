# 4. Explicit exceptions — bypass the bastion for specific hosts
#    Must come BEFORE "Host *" — SSH uses first-value-wins per directive
Host github.com
    ProxyJump none

# 5. Global defaults — fallback for every host except the bastion
Host * !%%vmIp=windows-vm-ip%%
    ProxyJump %%vmUser=vm-user%%@%%vmIp=windows-vm-ip%%
    IdentityFile ~/.ssh/%%sshKey=id_ed25519%%

    # Optional — prevent silently-dropped idle connections
    ServerAliveInterval 30
    ServerAliveCountMax 3

    # Optional — fail fast instead of waiting the default 2 minutes
    ConnectTimeout 5
